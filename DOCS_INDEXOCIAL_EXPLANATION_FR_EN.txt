2.2. DÉCODAGE HORS-THREAD AVEC ISOLATS DART (ISOLATE.RUN)
--------------------------------------------------------------------------------
Pour éviter tout gel de l'interface lors de la lecture des 9 004 ressources JSON,
le décodage est exécuté sur une fonction top-level pure sans capture d'état.

Extrait de code — Fonction Isolat Top-Level (`computeParseSocialResources`) :

```dart
List<SocialResource> parseSocialResourcesJson(String jsonStr) {
  try {
    if (jsonStr.trim().isEmpty) return [];
    final dynamic decoded = jsonDecode(jsonStr);
    if (decoded is! List) return [];
    final List<SocialResource> list = [];
    for (final item in decoded) {
      if (item is Map) {
        try {
          final map = Map<String, dynamic>.from(item);
          list.add(SocialResource.fromJson(map));
        } catch (_) {}
      }
    }
    return list;
  } catch (e) {
    debugPrint('PARSE ERROR: $e');
    return [];
  }
}

Future<List<SocialResource>> computeParseSocialResources(String jsonStr) {
  return Isolate.run(() => parseSocialResourcesJson(jsonStr));
}
```

2.3. ALGORITHME DE TRI ALPHABÉTIQUE PERSONNALISÉ (LETTRES -> CHIFFRES -> SYMBOLES)
--------------------------------------------------------------------------------
Le tri du répertoire et des favoris respecte la priorité spécifique demandée :
1. Lettres (`A-Z` et caractères accentués français `à`, `é`, `è`, `ê`, `ô`, `ç`, etc.) = Priorité 0.
2. Chiffres (`0-9`) = Priorité 1.
3. Symboles et caractères spéciaux (`#`, `@`, `-`, `(`, etc.) = Priorité 2.

Extrait de code — Comparateur Personnalisé (`_compareCustomAlphabetical`) :

```dart
int _getCharCategoryPriority(int codeUnit, String char) {
  // Lettres (A-Z, a-z, Accents français, œ, æ) : Priorité 0
  if ((codeUnit >= 65 && codeUnit <= 90) ||
      (codeUnit >= 97 && codeUnit <= 122) ||
      (codeUnit >= 192 && codeUnit <= 255) ||
      char == 'œ' || char == 'Œ' || char == 'æ' || char == 'Æ') {
    return 0; // LETTRES EN PREMIER
  }
  // Chiffres (0-9) : Priorité 1
  if (codeUnit >= 48 && codeUnit <= 57) {
    return 1; // CHIFFRES EN SECOND
  }
  // Symboles & Caractères spéciaux : Priorité 2
  return 2; // SYMBOLES EN DERNIER
}

int _compareCustomAlphabetical(String strA, String strB) {
  final normA = strA.trim().toLowerCase();
  final normB = strB.trim().toLowerCase();

  if (normA.isEmpty && normB.isEmpty) return 0;
  if (normA.isEmpty) return 1;
  if (normB.isEmpty) return -1;

  final minLen = min(normA.length, normB.length);
  for (int i = 0; i < minLen; i++) {
    final charA = normA[i];
    final charB = normB[i];
    if (charA != charB) {
      final prioA = _getCharCategoryPriority(normA.codeUnitAt(i), charA);
      final prioB = _getCharCategoryPriority(normB.codeUnitAt(i), charB);

      if (prioA != prioB) {
        return prioA.compareTo(prioB);
      }
      return charA.compareTo(charB);
    }
  }

  return normA.length.compareTo(normB.length);
}
```

Future<void> _exportFavoriteGroupPdf({required String title, required List<SocialResource> resources}) async {
  final pdf = pw.Document();

  pdf.addPage(
    pw.MultiPage(
      pageFormat: PdfPageFormat.a4,
      margin: const pw.EdgeInsets.all(32),
      header: (context) => pw.Column(
        crossAxisAlignment: pw.CrossAxisAlignment.start,
        children: [
          pw.Row(
            mainAxisAlignment: pw.MainAxisAlignment.spaceBetween,
            children: [
              pw.Text('INDEXOCIAL QC', style: pw.TextStyle(fontSize: 18, fontWeight: pw.FontWeight.bold, color: PdfColors.amber900)),
              pw.Text('Répertoire Civique du Québec', style: const pw.TextStyle(fontSize: 10, color: PdfColors.grey700)),
            ],
          ),
          pw.Divider(thickness: 1.5, color: PdfColors.amber700),
        ],
      ),
      build: (context) => [
        pw.Text(title.toUpperCase(), style: pw.TextStyle(fontSize: 15, fontWeight: pw.FontWeight.bold)),
        pw.SizedBox(height: 14),
        ...resources.map((r) {
          return pw.Container(
            margin: const pw.EdgeInsets.only(bottom: 10),
            padding: const pw.EdgeInsets.all(10),
            decoration: pw.BoxDecoration(
              border: pw.Border.all(color: PdfColors.grey400, width: 0.8),
              borderRadius: const pw.BorderRadius.all(pw.Radius.circular(6)),
            ),
            child: pw.Column(
              crossAxisAlignment: pw.CrossAxisAlignment.start,
              children: [
                pw.Text(r.name(_isFr), style: pw.TextStyle(fontSize: 12, fontWeight: pw.FontWeight.bold)),
                if (r.phone.isNotEmpty) pw.Text('Téléphone : ${r.phone}', style: const pw.TextStyle(fontSize: 9.5)),
                if (r.website.isNotEmpty) pw.Text('Site Web : ${r.website}', style: const pw.TextStyle(fontSize: 9.5)),
                if (r.region.isNotEmpty) pw.Text('Région : ${r.region}', style: const pw.TextStyle(fontSize: 9.5)),
              ],
            ),
          );
        }),
      ],
    ),
  );

  final bytes = await pdf.save();
  await Printing.sharePdf(bytes: bytes, filename: 'indexocial_favoris.pdf');
}

          pw.Divider(thickness: 1.5, color: PdfColors.amber700),
        ],
      ),
      build: (context) => [
        pw.Text(title.toUpperCase(), style: pw.TextStyle(fontSize: 15, fontWeight: pw.FontWeight.bold)),
        pw.SizedBox(height: 14),
        ...resources.map((r) {
          return pw.Container(
            margin: const pw.EdgeInsets.only(bottom: 10),
            padding: const pw.EdgeInsets.all(10),
            decoration: pw.BoxDecoration(
              border: pw.Border.all(color: PdfColors.grey400, width: 0.8),
              borderRadius: const pw.BorderRadius.all(pw.Radius.circular(6)),
            ),
            child: pw.Column(
              crossAxisAlignment: pw.CrossAxisAlignment.start,
              children: [
                pw.Text(r.name(_isFr), style: pw.TextStyle(fontSize: 12, fontWeight: pw.FontWeight.bold)),
                if (r.phone.isNotEmpty) pw.Text('Phone: ${r.phone}', style: const pw.TextStyle(fontSize: 9.5)),
                if (r.website.isNotEmpty) pw.Text('Website: ${r.website}', style: const pw.TextStyle(fontSize: 9.5)),
                if (r.region.isNotEmpty) pw.Text('Region: ${r.region}', style: const pw.TextStyle(fontSize: 9.5)),
              ],
            ),
          );
        }),
      ],
    ),
  );

  final bytes = await pdf.save();
  await Printing.sharePdf(bytes: bytes, filename: 'indexocial_favorites.pdf');
}

3. GRADLE BUILD SCRIPTS & ANDROID SIGNING (android/app/build.gradle.kts)
================================================================================
To automate renaming the Android release App Bundle (`.aab`) during compilation,
a custom Gradle `doLast` task is appended to `bundleRelease`.

Code Snippet — `android/app/build.gradle.kts`:

```kotlin
flutter {
    source = "../.."
}

tasks.whenTaskAdded {
    if (name == "bundleRelease") {
        doLast {
            val outputDir = file("${layout.buildDirectory.get()}/outputs/bundle/release")
            val defaultAab = file("${outputDir}/app-release.aab")
            val customAab = file("${outputDir}/com.iammithra.indexocial.aab")
            val projectRootAab = rootProject.file("../com.iammithra.indexocial.aab")
            if (defaultAab.exists()) {
                defaultAab.copyTo(customAab, overwrite = true)
                defaultAab.copyTo(projectRootAab, overwrite = true)
            }
        }
    }
}
```
